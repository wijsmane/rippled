125// extract first Memo STObject from applied STTx and check that it is valid
126// sfMemos is an STArray where each element contains an sfMemo
127static std::optional<STObject>
128firstMemoObject(std::shared_ptr<STTx const> const& tx)
129{
130 if (!tx || !tx->isFieldPresent(sfMemos))
131 return std::nullopt;
132
133 auto const& memosArr = tx->getFieldArray(sfMemos);
134 if (memosArr.empty())
135 return std::nullopt;
136
137 auto const& item = memosArr[0];
138
139 if (item.isFieldPresent(sfMemo))
140 return item.getFieldObject(sfMemo);
141
142 return item;
143}

145// Journal for deposit:
146// mode(1) || cm(32) => len = 33
147static std::optional<uint256>
148parseJournalDepositCm(Risc0Bytes const& journal)
149{
150 if (!journal.ptr || journal.len != 33)
151 return std::nullopt;
152
153 auto const* p = static_cast<std::uint8_t const*>(journal.ptr);
154 if (p[0] != kModeDeposit)
155 return std::nullopt;
156
157 uint256 cm;
158 std::memcpy(cm.begin(), p + 1, 32);
159 return cm;
160}

162// Journal for spend:
163// mode(1) || cm_old(32) || nf(32) => len = 65
164static std::optional<std::pair<uint256, uint256>>
165parseJournalSpendCmNf(Risc0Bytes const& journal)
166{
167 if (!journal.ptr || journal.len != 65)

172 return std::nullopt;

174 uint256 cmOld, nf;
