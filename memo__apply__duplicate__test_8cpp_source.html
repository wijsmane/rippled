24static uint256
25randomUint256()
26{
27 uint256 result;
28 std::random_device rd;
29 std::mt19937 gen(rd());
30 std::uniform_int_distribution<std::uint32_t> dis;
31
32 for (int i = 0; i < 8; ++i)
33 {
34 std::uint32_t v = dis(gen);
35 std::memcpy(result.begin() + i * 4, &v, 4);
36 }
37 return result;
38}

64append_u32_be(Blob& b, std::uint32_t v)
65{
66 b.push_back((v >> 24) & 0xFF);

72//build bind to make sure that even if someone copies cm,nf from memo,
73// the proof computation of cm is tied to the transaction
74static uint256
75makeBind(AccountID const& account, std::uint32_t seq)
76{
77 Blob b;
78 b.reserve(20 + 4);
79
80 b.insert(b.end(), account.begin(), account.end());
81
82 append_u32_be(b, seq);
83
84 return sha512Half(Slice(b.data(), b.size()));
85}

88// witness = amount(8) || rho(32) || r(32) || a_pk(32) || a_sk(32) || bind(32)
89static Blob
90pack_private(
91 std::uint64_t amount,
92 uint256 const& rho,
93 uint256 const& r,
94 uint256 const& a_pk,
95 uint256 const& a_sk,
96 uint256 const& bind)
97{
98 Blob out;
99 out.reserve(168);
100 ripple::zkp::append_u64_be(out, amount);
101 ripple::zkp::append_u256(out, rho);
102 ripple::zkp::append_u256(out, r);
103 ripple::zkp::append_u256(out, a_pk);
104 ripple::zkp::append_u256(out, a_sk);
105 ripple::zkp::append_u256(out, bind);
106 return out;
107}

109// retrieve cm and nf (public outputs) from journal
110static std::pair<uint256, uint256>
111parseJournalCmNf(Risc0Bytes const& journal)
112{
113 uint256 cm, nf;
114 if (!journal.ptr || journal.len != 64)
115 return {uint256{}, uint256{}};
116
117 std::memcpy(cm.begin(), journal.ptr, 32);
118 std::memcpy(nf.begin(), static_cast<std::uint8_t const*>(journal.ptr) + 32, 32);
119 return {cm, nf};
120}

122// attach memo with MemoType="zkp", MemoFormat="cmnf-v1", MemoData = hex(cm)||hex(nf)
123static void
124attachZkpMemo(test::jtx::JTx& jt, uint256 const& cm, uint256 const& nf)
125{
126 Json::Value memo(Json::objectValue);
127 memo[sfMemoType.jsonName] = strHex(std::string_view("zkp"));
128 memo[sfMemoFormat.jsonName] = strHex(std::string_view("cmnf-v1"));
129 memo[sfMemoData.jsonName] = hex256(cm) + hex256(nf); // have to convert to hex
130
131 Json::Value wrap(Json::objectValue);
132 wrap[sfMemo.jsonName] = memo;
133
134 Json::Value memos(Json::arrayValue);
135 memos.append(wrap);
136
137 jt.jv[sfMemos.jsonName] = memos;
138}

140// extract first Memo STObject from applied STTx and check that it is valid
141// sfMemos is an STArray where each element contains an sfMemo
142static std::optional<STObject>
143firstMemoObject(std::shared_ptr<STTx const> const& tx)
144{
145 if (!tx || !tx->isFieldPresent(sfMemos))
146 return std::nullopt;
147
148 auto const& memosArr = tx->getFieldArray(sfMemos);
149 if (memosArr.empty())
150 return std::nullopt;
151
152 auto const& item = memosArr[0];
153
154 if (item.isFieldPresent(sfMemo))
155 return item.getFieldObject(sfMemo); // copy out
156 return item; // copy out
157}

169 //create a funded account
170 Account alice{"alice"};
171 Env env(*this);

173 env.close(); //advance the ledger so the new state is finalised
174
175 //generate values for first transaction
176 uint256 const rho = randomUint256();
177 uint256 const a_sk = spendKeyFromString("spend_key_123456");
178
179 std::uint64_t const amount1 = 1000;
180 uint256 const r1 = randomUint256();
181 uint256 const a_pk1 = randomUint256();
182
183 std::cout<< "generated private inputs\n" <<std::endl;
184
185 //retrieve sequence number so we can bind to the current sequence
186 std::uint32_t const seq1 = currentAccountSequence(env, alice);
187 BEAST_EXPECT(seq1 != 0);
188
189 uint256 const bind1 = makeBind(alice.id(), seq1);
190
191 std::cout<<"bound inputs to this tx\n"<<std::endl;
192
193 //put all private inputs together to send to prover as the witness
194 Blob const witness1 = pack_private(amount1, rho, r1, a_pk1, a_sk, bind1);
195 BEAST_EXPECT(witness1.size() == 168);
196 std::cout<<"packed bytes to send to Risc0 prover, now generating proof\n"<<std::endl;
197
198 //generate proof
199 Risc0Bytes receipt1 = risc0_prove_zk_inputs(witness1.data(), witness1.size());
200 BEAST_EXPECT(receipt1.ptr && receipt1.len);
201 std::cout<<"proof generated successfully\n"<<std::endl;
202
203 //verify proof
204 int const vrc1 = risc0_verify_receipt(receipt1.ptr, receipt1.len);
205 BEAST_EXPECT(vrc1 == 0);
206 std::cout<<"receipt verified\n"<<std::endl;
207
208 //get public output
209 Risc0Bytes journal1 = risc0_receipt_get_journal(receipt1.ptr, receipt1.len);
210 BEAST_EXPECT(journal1.ptr != nullptr);
211 BEAST_EXPECT(journal1.len == 64);
212 std::cout << "journal processed successfully\n" << std::endl;

214 //get commitment and nullifier
215 auto const [cm1, nf] = parseJournalCmNf(journal1);
216 BEAST_EXPECT(cm1 != beast::zero);
217 BEAST_EXPECT(nf != beast::zero);

223 attachZkpMemo(jt, cm1, nf);
224 env(jt);
225 env.close();

252

312 if (journal2.ptr && journal2.len) risc0_free_bytes(const_cast<std::uint8_t*>(journal2.ptr), journal2.len);
313 if (receipt2.ptr && receipt2.len) risc0_free_bytes(const_cast<std::uint8_t*>(receipt2.ptr), receipt2.len);
314 }
