72// commitment = sha512Half( 0x01 || amount(8) || rho(32) || r(32) || a_pk(32) )
73inline uint256
74computeCommitment(ZkNote const& note)
75{
76 Blob buf;
77 buf.reserve(1 + 8 + 32 + 32 + 32);
78
79 buf.push_back(0x01);
80 append_u64_be(buf, note.amount);
81 append_u256(buf, note.rho);
82 append_u256(buf, note.r);
83 append_u256(buf, note.a_pk);
